% ==============================================================================
% 1. Resistor (IEEE zigzag default or IEC rectangular via 'iec resistor')
% ==============================================================================
\pgfdeclareshape{commtikz_resistor}{
  \commtikz@inheritrectanchors{commrect}

  \backgroundpath{%
    \commtikz@rectparams
    % Mask background wire behind resistor body
    \pgfscope
      \pgfsetfillcolor{\pgfkeysvalueof{/tikz/commtikz/fill}}%
      \pgfpathrectanglecorners{\pgfpoint{-0.70\commtikz@hw}{-0.75\commtikz@hh}}{\pgfpoint{0.70\commtikz@hw}{0.75\commtikz@hh}}%
      \pgfusepath{fill}%
    \endpgfscope
    \ifnum\pgfkeysvalueof{/tikz/commtikz/iec}=1\relax
      % IEC rectangular resistor
      \pgfpathmoveto{\pgfpoint{-\commtikz@hw}{0pt}}%
      \pgfpathlineto{\pgfpoint{-0.68\commtikz@hw}{0pt}}%
      \pgfpathrectanglecorners{\pgfpoint{-0.68\commtikz@hw}{-0.55\commtikz@hh}}{\pgfpoint{0.68\commtikz@hw}{0.55\commtikz@hh}}%
      \pgfpathmoveto{\pgfpoint{0.68\commtikz@hw}{0pt}}%
      \pgfpathlineto{\pgfpoint{\commtikz@hw}{0pt}}%
      \pgfusepath{stroke}%
    \else
      % IEEE 6-peak zigzag resistor
      \pgfpathmoveto{\pgfpoint{-\commtikz@hw}{0pt}}%
      \pgfpathlineto{\pgfpoint{-0.72\commtikz@hw}{0pt}}%
      \pgfpathlineto{\pgfpoint{-0.60\commtikz@hw}{0.70\commtikz@hh}}%
      \pgfpathlineto{\pgfpoint{-0.36\commtikz@hw}{-0.70\commtikz@hh}}%
      \pgfpathlineto{\pgfpoint{-0.12\commtikz@hw}{0.70\commtikz@hh}}%
      \pgfpathlineto{\pgfpoint{0.12\commtikz@hw}{-0.70\commtikz@hh}}%
      \pgfpathlineto{\pgfpoint{0.36\commtikz@hw}{0.70\commtikz@hh}}%
      \pgfpathlineto{\pgfpoint{0.60\commtikz@hw}{-0.70\commtikz@hh}}%
      \pgfpathlineto{\pgfpoint{0.72\commtikz@hw}{0pt}}%
      \pgfpathlineto{\pgfpoint{\commtikz@hw}{0pt}}%
      \pgfusepath{stroke}%
    \fi
    \ifnum\pgfkeysvalueof{/tikz/commtikz/tunable}=1\relax
      \commtikz@drawopenarrow{-0.65\commtikz@hw}{-0.95\commtikz@hh}{0.65\commtikz@hw}{0.95\commtikz@hh}%
    \fi
  }
}

% Numeric border anchors and labels must include the optical arrows, including
% after rotation. Port anchors still identify the electrical leads.
\def\commtikz@optodiodeborder{%
  \pgf@xa=\pgf@x\relax
  \pgf@ya=\pgf@y\relax
  \commtikz@rectparams
  \commtikz@phh=\commtikz@hh\relax
  \ifdim\pgf@ya>0pt
    \commtikz@phh=1.70\commtikz@hh\relax
  \fi
  \pgfpointborderrectangle{\pgfpoint{\pgf@xa}{\pgf@ya}}{\pgfpoint{\commtikz@hw}{\commtikz@phh}}%
}
