\newdimen\testactual
\newdimen\testexpected
\newcommand\assertdistance[5]{%
  \csname pgfextract#1\endcsname{\testactual}{\pgfpointdiff{\pgfpointanchor{#2}{#3}}{\pgfpointanchor{#2}{#4}}}%
  \testexpected=#5\relax
  \advance\testactual by -\testexpected
  \ifdim\testactual<0pt \testactual=-\testactual\fi
  \ifdim\testactual>0.02pt
    \errmessage{Unexpected #1 distance for #2.#3/#4}%
  \fi
}
\tikzset{assert stroke/.code={%
  \edef\testactualcolor{\tikz@strokecolor}%
  \def\testexpectedcolor{#1}%
  \ifx\testactualcolor\pgfutil@empty \def\testactualcolor{.}\fi
  \ifx\testexpectedcolor\pgfutil@empty \def\testexpectedcolor{.}\fi
  \convertcolorspec{named}{\testactualcolor}{rgb}\testactualrgb
  \convertcolorspec{named}{\testexpectedcolor}{rgb}\testexpectedrgb
  \ifx\testactualrgb\testexpectedrgb\else
    \errmessage{Expected stroke #1, got \testactualcolor}%
  \fi
}}
% Observe the head dimensions actually used while rendering, across all heads
% of a component (including optical emission pairs and curved arrows).
\let\originalarrowgeometry\commtikz@arrowgeometry
\def\assertcapturepath{%
  \pgfsyssoftpath@getcurrentpath\testpathbefore
  \tikzset{commtikz save path color}%
  \pgfsyssoftpath@getcurrentpath\testpathafter
  \ifx\testpathbefore\testpathafter\else
    \errmessage{Color capture changed the existing connector path}%
  \fi
}
\newif\ifcheckarrowheads
\newdimen\expectedheadlength
\newdimen\expectedheadwidth
\def\commtikz@arrowgeometry#1#2#3#4{%
  \originalarrowgeometry{#1}{#2}{#3}{#4}%
  \ifcheckarrowheads
    \testactual=\dimexpr\commtikz@arr@hl-\expectedheadlength\relax
    \ifdim\testactual<0pt \testactual=-\testactual\fi
    \ifdim\testactual>0.01pt \errmessage{Inconsistent arrowhead length}\fi
    \testactual=\dimexpr\commtikz@arr@hw-\expectedheadwidth\relax
    \ifdim\testactual<0pt \testactual=-\testactual\fi
    \ifdim\testactual>0.01pt \errmessage{Inconsistent arrowhead width}\fi
  \fi
}
\newcommand\assertgap[6]{%
  \csname pgfextract#1\endcsname{\testactual}{\pgfpointdiff{\pgfpointanchor{#2}{#3}}{\pgfpointanchor{#4}{#5}}}%
  \testexpected=#6\relax
  \advance\testactual by -\testexpected
  \ifdim\testactual<0pt \testactual=-\testactual\fi
  \ifdim\testactual>0.02pt \errmessage{Incorrect label clearance for #2/#4: offset \the\testactual}\fi
}
