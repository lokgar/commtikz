% Helper macro: Draw a smooth cubic Bezier sine wave centered at (0, #1) spanning [-0.6*hw, +0.6*hw]
\def\commtikz@drawsinewave#1{%
  \pgfmathsetlength\commtikz@sy{#1}%
  \pgfpathmoveto{\pgfpoint{-0.60\commtikz@hw}{\commtikz@sy}}%
  \pgfpathcurveto
    {\pgfpoint{-0.40\commtikz@hw}{\dimexpr\commtikz@sy + 0.27\commtikz@hh\relax}}
    {\pgfpoint{-0.20\commtikz@hw}{\dimexpr\commtikz@sy + 0.27\commtikz@hh\relax}}
    {\pgfpoint{0pt}{\commtikz@sy}}%
  \pgfpathcurveto
    {\pgfpoint{0.20\commtikz@hw}{\dimexpr\commtikz@sy - 0.27\commtikz@hh\relax}}
    {\pgfpoint{0.40\commtikz@hw}{\dimexpr\commtikz@sy - 0.27\commtikz@hh\relax}}
    {\pgfpoint{0.60\commtikz@hw}{\commtikz@sy}}%
}

% ==============================================================================
% 1. Amp & VGA (Triangle Amplifier / Variable Gain Amplifier / EDFA / SOA)
% ==============================================================================
\pgfdeclareshape{commtikz_amp}{
  \commtikz@inheritrectanchors{commrect}

  \savedanchor\amptextpoint{%
    \pgfmathsetlength\commtikz@u{\pgfkeysvalueof{/tikz/commtikz/unit}}%
    \pgfmathparse{\pgfkeysvalueof{/tikz/commtikz/width}}%
    \ifpgfmathunitsdeclared
      \pgfmathsetlength\commtikz@hw{0.5 * \pgfmathresult pt}%
    \else
      \pgfmathsetlength\commtikz@hw{0.5 * \pgfmathresult * \commtikz@u}%
    \fi
    \pgf@x=-0.30\commtikz@hw\relax
    \advance\pgf@x by -0.5\wd\pgfnodeparttextbox\relax
    \pgf@y=-0.5\ht\pgfnodeparttextbox\relax
    \advance\pgf@y by 0.5\dp\pgfnodeparttextbox\relax
  }
  \anchor{text}{\amptextpoint}

  % Slanted edge midpoint anchors for north/south
  \anchor{north}{\commtikz@rectparams\pgfpoint{0pt}{0.5\commtikz@hh}}
  \anchor{south}{\commtikz@rectparams\pgfpoint{0pt}{-0.5\commtikz@hh}}
  \anchor{N}{\commtikz@rectparams\pgfpoint{0pt}{0.5\commtikz@hh}}
  \anchor{S}{\commtikz@rectparams\pgfpoint{0pt}{-0.5\commtikz@hh}}
  \anchor{N0}{\commtikz@rectparams\pgfpoint{0pt}{0.5\commtikz@hh}}
  \anchor{S0}{\commtikz@rectparams\pgfpoint{0pt}{-0.5\commtikz@hh}}
  \anchor{tune}{%
    \commtikz@rectparams
    \ifnum\pgfkeysvalueof{/tikz/commtikz/tuneup}=1\relax
      \pgfpoint{-0.1333\commtikz@hw}{\commtikz@hh}%
    \else
      \pgfpoint{-0.1333\commtikz@hw}{-\commtikz@hh}%
    \fi
  }

  \backgroundpath{%
    \commtikz@rectparams
    \ifnum\pgfkeysvalueof{/tikz/commtikz/flip}=1\relax
      \pgfpathmoveto{\pgfpoint{\commtikz@hw}{-\commtikz@hh}}%
      \pgfpathlineto{\pgfpoint{-\commtikz@hw}{0pt}}%
      \pgfpathlineto{\pgfpoint{\commtikz@hw}{\commtikz@hh}}%
      \pgfpathclose
    \else
      \pgfpathmoveto{\pgfpoint{-\commtikz@hw}{-\commtikz@hh}}%
      \pgfpathlineto{\pgfpoint{\commtikz@hw}{0pt}}%
      \pgfpathlineto{\pgfpoint{-\commtikz@hw}{\commtikz@hh}}%
      \pgfpathclose
    \fi
  }

  \beforebackgroundpath{%
    \commtikz@rectparams
    \ifnum\pgfkeysvalueof{/tikz/commtikz/tunable}=1\relax
      \commtikz@drawarrow{-1.20\commtikz@hw}{-\commtikz@hh}{1.00\commtikz@hw}{\commtikz@hh}{0.16\commtikz@u}{0.11\commtikz@u}%
    \fi
    \commtikz@drawcomplabel{-0.30\commtikz@hw}{0pt}%
  }
}
